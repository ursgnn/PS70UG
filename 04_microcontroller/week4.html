int potPin = A0;    

int motorPin = D7;    


int potVal;

int motorSpeed;


const int freq = 5000;

const int resolution = 8; 


void setup() {

  Serial.begin(9600);


  pinMode(potPin, INPUT);

  pinMode(motorPin, OUTPUT);


  ledcAttach(motorPin, freq, resolution);

}


void loop() {

 

  potVal = analogRead(potPin);


  if (potVal < 100) {

    motorSpeed = 0;

  } else {

    motorSpeed = map(potVal, 100, 4095, 0, 255);

  }


  ledcWrite(motorPin, motorSpeed);


  Serial.print("Potentiometer: ");

  Serial.print(potVal);

  Serial.print("    Motor PWM: ");

  Serial.println(motorSpeed);


  delay(100);

}
